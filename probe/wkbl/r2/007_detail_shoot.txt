function shootData(coordinateX, coordinateY, order, type){ //X좌표, Y좌표, 순서, 슛성공여부(0 - 성공, 1 - 실패)
	coordinateX = Number(coordinateX);
	coordinateY = Number(coordinateY);

	if (coordinateX <= 520){
		coordinateX = coordinateX - 260;
	} else if (coordinateX <= 750){
		coordinateX = coordinateX - 230;
	} else if (coordinateX <= 900){
		coordinateX = coordinateX - 190;
	} else if (coordinateX <= 1000){
		coordinateX = coordinateX - 170;
	} else if (coordinateX <= 1160){
		coordinateX = coordinateX - 150;
	} else if (coordinateX <= 1540){
		coordinateX = coordinateX - 120;
	} else if(coordinateX <= 1990){
		coordinateX = coordinateX - 86;
	} else if(coordinateX <= 2140){
		coordinateX = coordinateX - 43;
	}

	if (coordinateY <= 900){
		coordinateY = coordinateY - 50;	
	} else {
		coordinateY = coordinateY + 30;	
	}

	var wrapCourt = '.wrap_court',
		pointX = coordinateX * 0.205673,
		pointY = coordinateY * 0.236842,
		pointType;

	pointX = Math.round(pointX);
	pointY = Math.round(pointY);

	if(type == '0'){
		pointType = "success";
	} else if (type == '1'){
		pointType = "fail";
	}

	$(wrapCourt).append('<span class="shooting_mark ' + pointType + '" style="top:' + pointY + 'px;left:' + pointX + 'px;" data-target="shootPoint' + order + '"></span>');
}

function uiShootChart(){
	var shootMark = '.shooting_mark',
		shootRow = '.chart_shooting .info_table01 tr';

	$(shootMark).mouseover(function(){
		var $dataTarget = $(this).attr('data-target'),
			$targetOffsetTop = $('#'+$dataTarget).offset().top - 50;
		$('#'+$dataTarget).addClass('on');

		if ($(window).scrollTop() > 956){

			if($targetOffsetTop > $(document).height()-1100){

				$('body, html').animate({
					scrollTop: $(document).height()-1085 + 'px'
				}, 0);
				
			} else {
				
				$('body, html').animate({
					scrollTop: $targetOffsetTop + 'px'
				}, 0);
			}
			
		}
	});

	$(shootMark).click(function(){
		var $dataTarget = $(this).attr('data-target'),
			$targetOffsetTop = $('#'+$dataTarget).offset().top - 50;
		$('#'+$dataTarget).addClass('on');
		if ($(window).scrollTop() < 956){
			$('body, html').animate({
				scrollTop: $targetOffsetTop + 'px'
			}, 1000);
		}
		
	});

	$(shootMark).mouseleave(function(){
		var $dataTarget = $(this).attr('data-target');
		$('#'+$dataTarget).removeClass('on');
	});

	$(shootRow).mouseover(function(){
		var $pointTarget = $(this).attr('id');
		$('.shooting_mark[data-target="'+$pointTarget+'"]').addClass('on');
	});
	$(shootRow).mouseleave(function(){
		var $pointTarget = $(this).attr('id');
		$('.shooting_mark[data-target="'+$pointTarget+'"]').removeClass('on');
	});

	$(window).on("scroll",function(){
		var $court = $('.court_container'),
			offset = $court.offset();
		 if ($(window).scrollTop() > 956){
			$court.addClass('fixed');

			if($(window).scrollTop() > $(document).height()-1085){
				$court.addClass('fixed_bottom');
			} else {
				$court.removeClass('fixed_bottom');
			}
		} else {
			$court.removeClass('fixed');
		}
	});
}

uiShootChart();
